## Cómo reproducir (Stellar CLI 23.2.1, identidades locales del seed)

```bash
NET=testnet; GOV=CBBYI45J…AL32; TRE=CACZWU3B…PATB
stellar contract invoke --network $NET --id $GOV --source-account raiz-admin -- tally --proposal_id 0
stellar contract invoke --network $NET --id $TRE --source-account raiz-admin -- execute_proposal --proposal_id 1
stellar contract invoke --network $NET --id $TRE --source-account raiz-admin -- execute_proposal --proposal_id 2
stellar contract invoke --network $NET --id $GOV --source-account raiz-r-centro-1 -- create_proposal \
  --proposer $(stellar keys address raiz-r-centro-1) --barrio_id ce4712…0001 \
  --description '"Luces para el parque infantil"' --amount 500000 \
  --recipient $(stellar keys address raiz-m-centro-1) --duration_days 3     # → id 3
stellar contract invoke --network $NET --id $GOV --source-account raiz-r-centro-1 -- vote \
  --resident $(stellar keys address raiz-r-centro-1) --proposal_id 3 --support true   # ×3 residentes
# ídem Norte → id 4, 2 votos
```
